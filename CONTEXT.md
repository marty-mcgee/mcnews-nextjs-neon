// Fetch collision records with filters
async function fetchCollisions(options = {}) {
    const {
        limit = 100,
        offset = 0,
        county = null,
        year = null,
        severity = null
    } = options;
    
    // Build filters
    const filters = {};
    if (county) filters.county_name = county;
    if (year) filters.collision_year = year;
    if (severity) filters.collision_severity = severity;
    
    const params = {
        resource_id: RESOURCE_ID,
        limit: limit,
        offset: offset,
        filters: JSON.stringify(filters)
    };
    
    try {
        const response = await axios.get(`${CKAN_API}/datastore_search`, { params });
        
        if (response.data.success) {
            return {
                records: response.data.result.records,
                total: response.data.result.total,
                limit: response.data.result.limit,
                offset: response.data.result.offset
            };
        } else {
            throw new Error(response.data.error?.message || 'Unknown error');
        }
    } catch (error) {
        console.error('API Error:', error.message);
        return null;
    }
}

// Fetch all records for a county with pagination
async function fetchAllCollisionsByCounty(county, limit = 100) {

    // First fetch recent collisions, then filter by distance
    const result = await fetchCollisions({ limit: 1000 });
    
    if (!result) return [];
    
    const filtered = result.records.filter(record => {
        if (!record.latitude || !record.longitude) return false;
        const distance = calculateDistance(lat, lon, record.latitude, record.longitude);
        return distance <= radiusMiles;
    });
    
    return filtered.map(parseCollisionRecord);
}

// Helper: Haversine distance calculation
function calculateDistance(lat1, lon1, lat2, lon2) {
    const R = 3959; // Earth radius in miles
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLon = (lon2 - lon1) * Math.PI / 180;
    const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
              Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
              Math.sin(dLon/2) * Math.sin(dLon/2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return R * c;
}

// Get collision statistics by county
async function getCountyStatistics(county) {
    const result = await fetchCollisions({ county: county, limit: 5000 });
    
    if (!result) return null;
    
    const stats = {
        county: county,
        totalCollisions: result.total,
        bySeverity: {
            fatal: result.records.filter(r => r.collision_severity === 'Fatal').length,
            injury: result.records.filter(r => r.collision_severity === 'Injury').length,
            propertyDamage: result.records.filter(r => r.collision_severity === 'Property Damage').length
        },
        topFactors: {},
        monthlyTrends: {}
    };
    
    // Analyze primary collision factors
    const factors = {};
    result.records.forEach(r => {
        if (r.primary_collision_factor) {
            factors[r.primary_collision_factor] = (factors[r.primary_collision_factor] || 0) + 1;
        }
    });
    stats.topFactors = Object.entries(factors)
        .sort((a, b) => b[1] - a[1])
        .slice(0, 5);
    
    return stats;
}

    async fetchDistrictData(district) {

        const url = `${this.baseURL}/d${district}/lcs/lcsStatusD${district.toString().padStart(2, '0')}.json`;

            logger.debug(`Fetching district ${district}: ${url}`);

                headers: { 'User-Agent': this.userAgent },
                timeout: 15000,

            });
            
            if (response.status === 404) {
                logger.info(`No lane closure data for District ${district}`);
                return null;
            }
            
            logger.info(`✓ District ${district}: ${response.data?.lcsClosures?.length || 0} closures`);
            return response.data;
            
        } catch (error) {
            const responseTime = Date.now() - startTime;
            
            await this.logAPIRequest({
                endpoint: url,
                district,
                responseTime,
                success: false,
                errorMessage: error.message
            });
            
            logger.error(`✗ District ${district} failed: ${error.message}`);
            return null;
        }
    }

    async logAPIRequest(logData) {
        const query = `
            INSERT INTO api_request_logs 
            (endpoint, district, response_time_ms, status_code, success, records_fetched, error_message, response_size_bytes)
            VALUES ($1, $2, $3, $4, $5, $6, $7, $8)

        try {
            await pool.query(query, [
                logData.endpoint,
                logData.district,
                logData.responseTime,
                logData.statusCode,
                logData.success,
                logData.recordsFetched || 0,
                logData.errorMessage,
                logData.responseSize
            ]);
        } catch (error) {
            logger.error('Failed to log API request:', error);
        }
    }

    async processClosureData(district, data) {
        if (!data || !data.lcsClosures || !Array.isArray(data.lcsClosures)) {
            return 0;
        }
        
        let processed = 0;
        const client = await pool.connect();
        
        try {
            await client.query('BEGIN');
            
            for (const closure of data.lcsClosures) {
                const sourceId = closure.lcsClosureID || 
                                `${district}_${closure.route}_${closure.startDate}_${closure.startTime}`;
                
                // Check if closure already exists
                const existing = await client.query(
                    'SELECT closure_id, times_seen FROM lane_closures WHERE source_id = $1',
                    [sourceId]
                );
                
                const closureData = this.normalizeClosureData(district, closure, sourceId);
                
                if (existing.rows.length > 0) {
                    // Update existing closure
                    await client.query(`
                        UPDATE lane_closures 
                        SET 
                            last_seen = NOW(),
                            times_seen = times_seen + 1,
                            status = $2,
                            description = $3,
                            lanes_affected = $4,
                            end_date = $5,
                            end_time = $6,

                        sourceId,
                        closureData.status,
                        closureData.description,
                        closureData.lanesAffected,
                        closureData.endDate,
                        closureData.endTime,
                        JSON.stringify(closure)
                    ]);
                    
                    logger.debug(`Updated closure ${sourceId}`);
                } else {
                    // Insert new closure
                    await client.query(`
                        INSERT INTO lane_closures (
                            source_id, district, route, direction, closure_type, 
                            lanes_affected, start_date, end_date, start_time, end_time,
                            description, latitude, longitude, geom, status, raw_data

                        closureData.latitude,
                        closureData.longitude,
                        closureData.status,

                    ]);
                    
                    logger.info(`New closure ${sourceId} in District ${district}`);
                }
                
                processed++;
            }
            
            await client.query('COMMIT');
            
        } catch (error) {
            await client.query('ROLLBACK');
            logger.error(`Error processing district ${district}:`, error);
            throw error;
        } finally {
            client.release();
        }
        
        return processed;
    }

    normalizeClosureData(district, closure, sourceId) {
        return {
            sourceId,
            route: closure.route || 'Unknown',
            direction: closure.direction || 'Unknown',
            closureType: closure.closureType || closure.type || 'Unknown',
            lanesAffected: closure.lanesAffected || closure.lanesClosed || 'Unknown',
            startDate: closure.startDate || new Date().toISOString().split('T')[0],
            endDate: closure.endDate || '2099-12-31',
            startTime: closure.startTime || '00:00',
            endTime: closure.endTime || '23:59',
            description: closure.description || closure.comments || `${closure.closureType || 'Closure'} on ${closure.route}`,
            latitude: parseFloat(closure.latitude) || null,
            longitude: parseFloat(closure.longitude) || null,
            status: this.determineStatus(closure),
            rawData: closure
        };
    }

    determineStatus(closure) {
        // If end date is in the past, mark as completed
        if (closure.endDate) {
            const endDate = new Date(closure.endDate);
            if (endDate < new Date()) {
                return 'completed';
            }
        }
        return closure.status === 'cancelled' ? 'cancelled' : 'active';
    }

    async takeSnapshot() {
        logger.info('Taking hourly snapshot...');
        
        const query = `
            INSERT INTO lane_closures_snapshots (district, total_closures, closures_by_type, closures_by_route, raw_summary)
            SELECT 
                district,
                COUNT(*) as total_closures,
                JSONB_AGG(DISTINCT closure_type) as closures_by_type,
                JSONB_AGG(DISTINCT route) as closures_by_route,
                JSONB_BUILD_OBJECT(
                    'active_count', COUNT(CASE WHEN status = 'active' THEN 1 END),
                    'avg_lanes_affected', AVG(CASE WHEN lanes_affected ~ '^[0-9]+$' THEN lanes_affected::INTEGER ELSE NULL END)
                ) as raw_summary
            FROM lane_closures
            WHERE status = 'active'
            GROUP BY district
        `;
        
        try {
            const result = await pool.query(query);
            logger.info(`Snapshot saved: ${result.rowCount} districts recorded`);
        } catch (error) {
            logger.error('Snapshot failed:', error);
        }
    }

    async cleanupOldData() {
        logger.info('Running cleanup routine...');
        
        // Archive closures older than 7 days
        const archiveQuery = `
            UPDATE lane_closures 
            SET status = 'completed' 
            WHERE status = 'active' 
            AND end_date < CURRENT_DATE - INTERVAL '7 days'
            RETURNING closure_id
        `;
        
        const result = await pool.query(archiveQuery);
        
        if (result.rowCount > 0) {
            logger.info(`Archived ${result.rowCount} old closures`);
        }
        
        // Delete API logs older than 30 days
        const deleteLogs = await pool.query(
            "DELETE FROM api_request_logs WHERE request_timestamp < NOW() - INTERVAL '30 days'"
        );
        logger.info(`Cleaned up ${deleteLogs.rowCount} old API logs`);
    }

    async pollAllDistricts() {
        logger.info(`Starting poll cycle for ${this.districts.length} districts at ${new Date().toISOString()}`);
        
        let totalClosures = 0;
        const results = {};
        
        for (const district of this.districts) {
            // Add delay between districts to be respectful
            await new Promise(resolve => setTimeout(resolve, 500));
            
            const data = await this.fetchDistrictData(district);
            if (data) {
                const processed = await this.processClosureData(district, data);
                totalClosures += processed;
                results[district] = processed;
            }
        }
        
        logger.info(`Poll cycle complete: ${totalClosures} total closures processed`);
        logger.debug('District breakdown:', results);
        
        return { totalClosures, results };
    }

    async getActiveClosures(filters = {}) {
        let query = `
            SELECT 
                closure_id, district, route, direction, closure_type,
                lanes_affected, description, latitude, longitude,
                start_date, end_date, status, last_seen,
                EXTRACT(EPOCH FROM (end_timestamp - NOW()))/3600 as hours_remaining
            FROM lane_closures
            WHERE status = 'active'
        `;
        
        const params = [];
        let paramIndex = 1;
        
        if (filters.district) {
            query += ` AND district = $${paramIndex++}`;
            params.push(filters.district);
        }
        
        if (filters.route) {
            query += ` AND route ILIKE $${paramIndex++}`;
            params.push(`%${filters.route}%`);
        }
        
        if (filters.county) {
            query += ` AND county = $${paramIndex++}`;
            params.push(filters.county);
        }
        
        query += ' ORDER BY end_date ASC, start_date ASC';
        
        const result = await pool.query(query, params);
        return result.rows;
    }
}

            params.push(`%${route}%`);
        }
        
        if (county) {
            query += ` AND lc.county ILIKE $${paramCount++}`;
            params.push(`%${county}%`);
        }
        
        query += ` ORDER BY lc.end_date ASC LIMIT $${paramCount}`;
        params.push(limit);
        
        const result = await pool.query(query, params);
        return result.rows;
    }

        const query = `
            SELECT 
                DATE(start_timestamp) as date,
                COUNT(*) as total_closures,
                COUNT(CASE WHEN status = 'active' THEN 1 END) as active_count,
                ARRAY_AGG(DISTINCT closure_type) as types
            FROM lane_closures
            WHERE route ILIKE $1
                AND start_timestamp > NOW() - INTERVAL '${days} days'
            GROUP BY DATE(start_timestamp)
            ORDER BY date DESC
        `;
        
        const result = await pool.query(query, [route]);
        return result.rows;
    }

                dd.district_name,

                MIN(lc.end_date) as earliest_end
            FROM caltrans_districts dd
            LEFT JOIN lane_closures lc ON dd.district_id = lc.district 
                AND lc.status = 'active'
            GROUP BY dd.district_id, dd.district_name, dd.region
            ORDER BY active_closures DESC
        `;

export class CHPPoller {

  async fetchCollisions(options: {

    endDate?: string;
  } = {}) {
    const startTime = Date.now();

    try {
      // Build filters for date range
      const filters: any = {};
      if (startDate) {
        filters['Crash Date Time'] = `>= ${startDate}`;
      }
      if (endDate) {
        filters['Crash Date Time'] = filters['Crash Date Time']

      }
      
      const params: any = {
        resource_id: this.resourceId,
        limit,
        offset
      };
      
      if (Object.keys(filters).length > 0) {

  private async upsertCollision(record: any): Promise<'new' | 'updated' | 'skipped'> {
    const caseId = record['Report Number'];
    
    if (!caseId) {
      return 'skipped';
    }
    
    let collisionDate: Date | null = null;
    let collisionYear: number | null = null;
    if (record['Crash Date Time']) {

      collisionYear = collisionDate.getFullYear();
    }

      collisionYear: collisionYear,

      location: this.buildLocation(record),
      latitude: record['Latitude'] ? parseFloat(record['Latitude']) : null,
      longitude: record['Longitude'] ? parseFloat(record['Longitude']) : null,
      primaryFactor: record['Primary Collision Factor Violation'],
      weather: record['Weather 1'],
      lighting: record['LightingDescription'],
      injuries: record['NumberInjured'] || 0,
      fatalities: record['NumberKilled'] || 0,
      rawData: record,
      lastSeen: new Date(),
    };
    
    try {
      const existing = await db
        .select()
        .from(chpCollisions)
        .where(eq(chpCollisions.caseId, caseId))
        .limit(1);
      
      if (existing.length > 0) {
        return 'skipped';

        return 'new';

      return 'skipped';

    if (this.pollingActive) {
      return { success: false, error: 'Polling already in progress' };
    }
    
    this.pollingActive = true;
    const startTime = Date.now();
    
    try {

      let offset = 0;

        newCount,

        skippedCount,

        dateRange: { startDate, endDate }
      };
      
      console.log(`✅ CHP Historical Poll complete:`);
      console.log(`  ${allRecords.length} fetched, ${newCount} new, ${updatedCount} updated, ${skippedCount} skipped`);
      console.log(`  Duration: ${duration}ms`);
      
      return {
        success: true,
        stats: this.lastPollStats,
        timestamp: new Date().toISOString()
      };
      
    } catch (error) {
      console.error('CHP Historical Polling error:', error);
      return { success: false, error: String(error) };
    } finally {
      this.pollingActive = false;
    }
  }

  async getStats() {

## 🚦 Caltrans CWWP2 Poller (Real-time Lane Closures)

**Source:** `https://cwwp2.dot.ca.gov/data/d{1-12}/lcs/lcsStatusDXX.json`  
**Format:** JSON (official, no authentication)  
**Update frequency:** Every 5 minutes (data refreshes)  
**Polling interval:** Configurable (default 5 min)

### Features
- Polls all 12 Caltrans districts sequentially
- Upserts closures by unique `source_id`
- Marks closures as `completed` if not seen for 15 minutes
- Stores raw JSON in `raw_data` column
- API logging for monitoring

### API Endpoints

GET /api/caltrans/poll - Trigger immediate poll
GET /api/caltrans/cron/poll - For scheduled polling
GET /api/caltrans/closures - Query closures (supports district, route, status)
GET /api/caltrans/closures/stats - Get polling statistics
GET /api/caltrans/closures/export - Export to CSV/JSON

📊 CHP Historical Poller (Working)

Source: data.ca.gov/api/3/action/datastore_search
Resource ID: b8ce0ca4-b4e9-490d-b4d1-1f4ec48cbefb
Format: JSON (CKAN API, no auth)
Data: Historical collision records (not real-time)
Key Implementation Details

    Pagination: Fetches 100 records at a time, loops until limit reached

    Date filtering: Client-side by Crash Date Time (CKAN doesn't support date operators in filters)

    Year filtering: Can filter for specific year (e.g., 2026 only)

    Duplicate detection: Checks caseId (Report Number) before insert

    Batch imports: Supports limit parameter (up to 5000+ records)

⚠️ Critical Decisions (From Development Chats)

    NO HTML scraping – The live CHP CAD page (cad.chp.ca.gov/Traffic.aspx) returns HTML, not JSON. We use official APIs instead.

    CKAN date filtering limitation – The CKAN API does NOT support >=, <=, or BETWEEN operators in filters. Solution: fetch records and filter client-side.

    All pollers use Drizzle ORM – No raw SQL or pg pool. Neon handles the connection.

    Vercel serverless functions – Can run up to 300 seconds (maxDuration = 300), suitable for batch imports.

    Batch imports for large datasets – For 120k+ records, import in chunks by date range or use limit.
